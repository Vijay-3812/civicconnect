      el.addEventListener('dragstart', (e) => { dragId = el.dataset.id; el.classList.add('dragging'); e.dataTransfer.effectAllowed = 'move'; try { e.dataTransfer.setData('text/plain', dragId); } catch (err) { /* Safari */ } });
      el.addEventListener('dragend', () => { el.classList.remove('dragging'); });
      el.addEventListener('keydown', (e) => {
        if (e.key === 'Enter') Router.go('#/c/' + el.dataset.id);
      });
    });
    Utils.$$('.kcol', root).forEach((col) => {
      col.addEventListener('dragover', (e) => {
        e.preventDefault();
        const target = col.dataset.col;
        const c = Store.complaint(dragId);
        const ok = c && TRANSITIONS[c.status] && TRANSITIONS[c.status][target];
        col.classList.toggle('drop-ok', !!ok); col.classList.toggle('drop-bad', !ok);
      });
      col.addEventListener('dragleave', () => { col.classList.remove('drop-ok', 'drop-bad'); });
      col.addEventListener('drop', (e) => {
        e.preventDefault();
        col.classList.remove('drop-ok', 'drop-bad');
        const target = col.dataset.col;
        const c = Store.complaint(dragId);
        if (!c) return;
        if (target === 'Resolved') { openProof(c); return; }
        const chk = Store.can(target === 'In Progress' ? 'progress' : target === 'Assigned' ? 'assign' : 'verify', c);
        if (!chk.ok) { shake(c.id); Toast.show(t('toast.roleDenied'), 'err'); return; }
        if (target === 'Assigned' && c.status === 'Verified') { openAssign(c); return; }
        const res = Actions.transition(c.id, target, { by: 'official', note: 'Moved on the department board' });
        if (!res.ok) shake(c.id);
      });
    });
    function shake(id) {
      const card = Utils.$('[data-id="' + id + '"]', root);
      if (card) { card.classList.add('shake'); setTimeout(() => card.classList.remove('shake'), 500); }
    }
  }
  const update = () => render(Utils.$('#viewRoot'));
  return { id: 'official', render, update };
};
