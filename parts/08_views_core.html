      const name = mode === 'register' && Utils.$('#aName') ? Utils.$('#aName').value.trim() : '';
      const res = mode === 'register' ? Store.Auth.register({ name, email, password: pass }) : Store.Auth.login({ email, password: pass });
      // Inline field errors (validation lives in the service layer, surfaced here)
      Utils.$('#eEmail').textContent = res.ok ? '' : (res.error === 'auth.err.email' ? t('auth.err.email') : '');

        draft.title = Utils.$('#rTitle').value.trim(); draft.description = Utils.$('#rDesc').value.trim();
        const eT = Utils.$('#eTitle'), eD = Utils.$('#eDesc');
        eT.textContent = draft.title.length < 8 ? t('form.tooShort', { n: 8 }) : '';
        eD.textContent = draft.description.length < 20 ? t('form.tooShort', { n: 20 }) : '';
        Utils.$('#rTitle').setAttribute('aria-invalid', String(draft.title.length < 8));
        Utils.$('#rDesc').setAttribute('aria-invalid', String(draft.description.length < 20));
        if (eT.textContent || eD.textContent) { Toast.show(t('form.required'), 'warn'); return; }
        runAnalysis(false); step = 1; render(Utils.$('#viewRoot')); return;
      }
      if (step === 1) {
        if (!draft.category) { Toast.show(t('form.needCat'), 'warn'); return; }
        step = 2; render(Utils.$('#viewRoot')); return;
      }
      if (step === 2) { step = 3; render(Utils.$('#viewRoot')); return; }
      if (step === 3) {
        if (!draft.location || !draft.location.lat) { Toast.show(t('form.needLoc'), 'warn'); return; }
        step = 4; render(Utils.$('#viewRoot')); return;
      }
      if (step === 4) { submitReport(false); }
    });
  }
  function mount() { if (step === 3) mountPicker(); if (step === 4) { runAnalysis(false); renderDupCard(); } }
  void ReportApi;
  function destroy() { destroyPicker(); if (recog && listening) { try { recog.stop(); } catch (e) { /* noop */ } } }
  /* ---------- API for the guided demo ---------- */
  ReportApi = {
    set(title, description, category, address) {
      draft.title = title; draft.description = description; draft.category = category;
      if (address) draft.location.address = address;
      const tIn = Utils.$('#rTitle'), dIn = Utils.$('#rDesc');
      if (tIn) tIn.value = title; if (dIn) dIn.value = description;
      runAnalysis();
    },
    gotoStep(n) { step = n; destroyPicker(); render(Utils.$('#viewRoot')); },
    getStep: () => step,
    analyze: () => runAnalysis(true),
    hasDuplicate: () => !!(analysis && analysis.duplicates.candidate),
    duplicateTitle: () => (analysis && analysis.duplicates.candidate ? analysis.duplicates.candidate.complaint.title : ''),
    submitInstead: (support) => submitReport(support),
    simulateMedia() {
      draft.media = [Utils.placeholderImage('demo-photo', 'pothole photo')];
      const grid = Utils.$('#mediaGrid'); if (grid) grid.innerHTML = draft.media.map((s, i) => mediaItem(s, i)).join('');
      Utils.$('#scanWrap') && Utils.$('#scanWrap').classList.remove('hide');
      runAnalysis(true);
    },
    setPoint,
    draft, get analysis() { return analysis; }
  };
  return { id: 'report', render, mount, destroy, update: () => {} };
};
